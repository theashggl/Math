import 'package:flutter/cupertino.dart';

class MyParentClass extends StatefulWidget {
  const MyParentClass({Key? key}) : super(key: key);

  @override
  State<MyParentClass> createState() => _MyParentClassState();
}

class _MyParentClassState extends State<MyParentClass> {
  MyState stateAboveChildren = MyState(name: 'name', age: 20);

  void _nameChanged(String value) {
    setState(() {
      stateAboveChildren = stateAboveChildren.copyWith(name: value);
    });
  }

  void _ageChanged(int value) {
    setState(() {
      stateAboveChildren = stateAboveChildren.copyWith(age: value);
    });
  }

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        ChildOne(state: stateAboveChildren, onChanged: _nameChanged),
        ChildTwo(state: stateAboveChildren, onChanged: _ageChanged),
      ],
    );
  }
}

class ChildOne extends StatelessWidget {
  final ValueChanged<String> onChanged;
  final MyState state;
  const ChildOne({Key? key, required this.state, required this.onChanged})
      : super(key: key);

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        TextFormField(
          initialValue: state.name,
          onChanged: onChanged,
        ),
        Text(state.name),
        Text('${state.age}')
      ],
    );
  }
}

class ChildTwo extends StatelessWidget {
  final ValueChanged<int> onChanged;
  final MyState state;
  const ChildTwo({Key? key, required this.state, required this.onChanged})
      : super(key: key);

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        TextFormField(
          initialValue: '${state.age}',
          keyboardType: const TextInputType.numberWithOptions(decimal: false),
          onChanged: (value) =>onChanged(int.parse(value)),
        ),
        Text('${state.age}'),
        Text(state.name),
      ],
    );
  }
}

class MyState {
  final String name;
  final int age;
  MyState({
    required this.name,
    required this.age,
  });

  MyState copyWith({
    String? name,
    int? age,
  }) {
    return MyState(
      name: name ?? this.name,
      age: age ?? this.age,
    );
  }
}
